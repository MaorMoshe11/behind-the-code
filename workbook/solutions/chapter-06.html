JOIN רגיל מחזיר רק שיחות שיש להן הודעה. LEFT JOIN שומר כל שורה של הטבלה השמאלית, וכשאין התאמה, העמודות של m הן NULL. לכן WHERE m.id IS NULL משאיר בדיוק את השיחות הריקות. (ב-SQL בודקים IS NULL, לא = NULL: NULL לא שווה לשום דבר, גם לא לעצמו.)

משימה 4: הניסיון הראשון נכשל, כי להודעה של c2 יש foreign key שמצביע על השיחה. מחיקה שלה הייתה משאירה הודעה יתומה, ובסיס הנתונים סירב. ה-PRAGMA נחוץ שוב, כי זה shell חדש, כלומר חיבור חדש. הסדר הנכון: קודם ההודעות, אחר כך השיחה, ב-transaction אחת, כדי שאף אחד לא יראה שיחה בלי הודעות באמצע. זה בדיוק מה ש-Conversations.delete עושה.

הבדלים ב-CREATE TABLE (כל שלושה מאלה מספיקים):

  • SQLAlchemy כתב VARCHAR(32) ו-VARCHAR(16) במקום TEXT, מ-String(32). SQLite לא אוכף את האורך, Postgres כן.
  • יש עמודה created_at DATETIME NOT NULL גם בהודעות.
  • ה-primary key וה-foreign key כתובים כשורות נפרדות בסוף (PRIMARY KEY (id), FOREIGN KEY(...) REFERENCES ...), ולא ליד העמודה. המשמעות זהה.
  • ה-index לא מופיע בטבלה: הוא פקודה נפרדת, CREATE INDEX ix_messages_conversation_id, שרואים ב-SELECT name, sql FROM sqlite_master WHERE type = 'index';.

3כמה שווה index

scripts\index_lab.py
"""How much does an index help? Times one query before and after CREATE INDEX.

Run it with: uv run python scripts/index_lab.py
Everything happens in an in-memory SQLite database, so no file is left behind.
"""

import sqlite3
import time

CONVERSATIONS = 10_000
MESSAGES_PER_CONVERSATION = 50
QUERY = "SELECT content FROM messages WHERE conversation_id = ? ORDER BY id"


def fill(db: sqlite3.Connection) -> None:
    """Create the messages table and fill it with fake messages."""
    db.execute(
        "CREATE TABLE messages (id INTEGER PRIMARY KEY, conversation_id TEXT NOT NULL,"
        " role TEXT NOT NULL, content TEXT NOT NULL)"
    )
    rows = (
        (f"c{c}", "user" if m % 2 == 0 else "assistant", f"message {m} of conversation {c}")
        for c in range(CONVERSATIONS)
        for m in range(MESSAGES_PER_CONVERSATION)
    )
    # The ? placeholders: the values travel separately from the SQL text.
    db.executemany("INSERT INTO messages (conversation_id, role, content) VALUES (?, ?, ?)", rows)
    db.commit()


def time_queries(db: sqlite3.Connection, label: str, repeats: int = 200) -> None:
    """Run the query for `repeats` different conversations and print the average time."""
    plan = db.execute("EXPLAIN QUERY PLAN " + QUERY, ("c1",)).fetchall()
    start = time.perf_counter()
    for i in range(repeats):
        db.execute(QUERY, (f"c{i * 37 % CONVERSATIONS}",)).fetchall()
    average_ms = (time.perf_counter() - start) / repeats * 1000
    print(f"{label}: {average_ms:.3f} ms per query  plan: {plan[0][3]}")


def main() -> None:
    db = sqlite3.connect(":memory:")
    fill(db)
    count = db.execute("SELECT COUNT(*) FROM messages").fetchone()[0]
    print(f"{count:,} messages in {CONVERSATIONS:,} conversations")
    time_queries(db, "without index")
    start = time.perf_counter()
    db.execute("CREATE INDEX ix_messages_conversation_id ON messages (conversation_id)")
    print(f"CREATE INDEX took {(time.perf_counter() - start) * 1000:.0f} ms")
    time_queries(db, "with index   ")
    db.close()


if __name__ == "__main__":
    main()

ה-generator rows חוסך לבנות רשימה של חצי מיליון tuples בזיכרון, ו-executemany עם commit אחד בסוף שולח הכול ב-transaction אחת. בקובץ אמיתי, commit אחרי כל שורה היה איטי בהרבה, כי כל COMMIT מחכה שהדיסק יאשר את הכתיבה (זה ה-durability מסעיף 6.6).

def chat_loop(client: ChatClient, settings: Settings, resume: bool = False) -> None:
    """Run the conversation until the user quits. `resume` continues the newest one."""
    print("Type a message and press Enter. Commands: /reset, /history, exit.\n")
    engine = make_engine(settings.database_url)
    conversations = Conversations(client, settings.max_turns, engine)
    # The terminal holds one conversation: the newest one, or a new one.
    last_id = conversations.latest() if resume else None
    if last_id is None:
        conversation_id = conversations.start()
    else:
        conversation_id = last_id
        turns = len(conversations.messages(conversation_id)) // 2
        print(f"Continuing the last conversation ({turns} turns).\n")

class BrokenClient:
    def reply(self, messages):
        raise LLMError("Could not reach the API. Check your internet connection.")


def count_messages(engine):
    """Ask the database directly, with SQL, instead of trusting the code under test."""
    with engine.connect() as connection:
        return connection.execute(text("SELECT COUNT(*) FROM messages")).scalar_one()


def test_a_failed_reply_saves_nothing():
    engine = make_engine("sqlite://")
    conversations = Conversations(BrokenClient(), 10, engine)
    conversation_id = conversations.start()
    with pytest.raises(LLMError):
        conversations.send(conversation_id, "hello")
    assert count_messages(engine) == 0


def test_delete_removes_the_messages_too():
    engine = make_engine("sqlite://")
    conversations = Conversations(MockClient(), 10, engine)
    keep = conversations.start()
    conversations.send(keep, "keep me")
    gone = conversations.start()
    conversations.send(gone, "delete me")
    conversations.delete(gone)
    assert count_messages(engine) == 2
    with pytest.raises(ConversationNotFound):
        conversations.messages(gone)


def test_the_database_refuses_a_message_without_a_conversation():
    engine = make_engine("sqlite://")
    with pytest.raises(IntegrityError), Session(engine) as session, session.begin():
        session.add(MessageRow(conversation_id="nope", role="user", content="orphan"))

6לבחור מאגר

אין תשובה יחידה. כך היינו עונים:

תרחישבחירהלמה, ומה המחיר
גרף יומי על 300 מיליון הודעותמתחילים בשאלה על Postgres עם index על התאריך. כשהיא לוקחת דקות: מאגר אנליטי (ClickHouse), שמקבל עותק של ההודעותסיכומים על עמודות מעטות מתוך המון שורות הם בדיוק מה שמאגר columnar טוב בו. המחיר: עוד מערכת, ותהליך שמעתיק אליה נתונים בלי לאבד ובלי לשכפל (פרק 12). המקור נשאר Postgres
תוצאות כלים ב-JSON משתנהעמודת JSON (ב-Postgres: JSONB) בטבלה של ההודעות, או טבלה tool_calls עם foreign key להודעההנתונים קשורים להודעה ונקראים איתה. מאגר מסמכים נפרד היה מפרק את מה שצריך להיות יחד, ומוותר על ה-transaction שכותבת את שניהם
אותה שאלה אלף פעם ביוםCache (Redis): מפתח מהשאלה המנורמלת, ערך התשובה, TTL של שעותקריאות חוזרות, ומותר לאבד את העותק: אם הוא נמחק, שואלים את Claude שוב. המחיר: תשובה ישנה עד שה-TTL פג, והחלטה מה נחשב "אותה שאלה". (ה-prompt caching של Anthropic, בפרק 19, הוא דבר אחר: הוא מוזיל חזרה על אותו context, לא מדלג על הקריאה)
60 בקשות בדקה לכל מפתחRedis: מונה לכל מפתח ודקה (INCR, ו-TTL של דקה)כתיבות רבות וקטנות, שאיבוד שלהן לא מזיק. כשיש שרת אחד, אפשר גם מונה בזיכרון. כשיש כמה, צריך מקום משותף, ובסיס נתונים רלציוני היה עמוס סתם
