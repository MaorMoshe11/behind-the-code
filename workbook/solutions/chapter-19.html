4לשכוח לפי בקשה

agent\memory_tools.py (added)
class ForgetInput(ToolInput):
    memory: str = Field(
        pattern=r"^[0-9a-f]{8}$",
        description="The memory's ref: the 8 characters in brackets that recall shows.",
    )


def forget_memory(context: ToolContext, args: ForgetInput) -> str:
    # Only this owner's memories are searched: a ref of someone else's is "not found".
    fact = context.memory_store().forget(args.memory)
    if fact is None:
        raise ToolError(f"No memory [{args.memory}]. Call recall to see the memories and refs.")
    return f"Forgotten: [{ref(fact)}] {fact.text}"

ורישום ב-memory_tools() עם writes=True, needs="memory"; ב-llm.py, UNREMEMBER = re.compile(r"\bforget memory \[?([0-9a-f]{8})\]?", re.IGNORECASE) ותנאי ב-wanted_calls. הבדיקה, ב-tests\test_memory.py:

def test_the_model_can_forget_only_this_clients_memories():
    conversations = Conversations(MockClient(), 10, make_engine("sqlite://"), write_tools=True)
    dana_fact, _ = conversations.memory("dana").remember("likes tea")
    omer = conversations.start("omer")
    refused = conversations.send(omer, f"Forget memory {dana_fact.id[:8]}.")

    assert "Forgotten:" in done

agent\memory.py
def stem(word: str) -> str:
    """A plural without its s, so that "cakes" also finds "cake" (a search for "cake" finds
    "cakes" already: it is contained in it). Crude, English only, and enough here."""
    if len(word) >= 4 and word.endswith("s") and not word.endswith("ss"):
        return word[:-1]
    return word
