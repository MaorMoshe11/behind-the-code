ההיגיון: ה-SDK מנסה שוב רק כשניסיון נוסף עשוי להצליח. 429 (חכו) ו-5xx (השרת עמוס או נפל) הם מצבים זמניים. 400 ו-401 לא: אותה בקשה, עם אותו מפתח, תיכשל שוב בדיוק באותה צורה, ולשלוח אותה שוב רק מבזבז זמן. בקוד של ה-SDK (_should_retry, פרק 3 תרגיל 5) הרשימה היא 408, 409, 429 וכל 5xx, ובנוסף שגיאות חיבור. בין ניסיון לניסיון ההמתנה גדלה (כחצי שנייה, ואז כשנייה): זה exponential backoff, כדי לא להציף שרת שכבר עמוס.

למה 401 סוגר את הסוכן: ב-llm.py, AuthenticationError הופכת ל-LLMError עם fatal=True, ו-chat_loop קורא ל-sys.exit(1). אותו היגיון כמו של ה-SDK, ברמה של הסוכן: מפתח שגוי לא יתקן את עצמו. שאר הקודים הם APIStatusError רגילה, והשיחה ממשיכה.

6probe.py: איפה זה נשבר?

פתרון אפשרי אחד. שלכם יכול להיראות אחרת לגמרי, כל עוד הוא עוצר בשלב הראשון שנכשל ואומר מה כנראה הבעיה.

scripts\probe.py
"""Find out at which step a request to a URL fails: DNS, connection, or HTTP.

Run it with: uv run python scripts/probe.py http://127.0.0.1:8000/
"""

import socket
import sys
from http.client import HTTPConnection, HTTPSConnection
from urllib.parse import urlsplit


def probe(url: str) -> bool:
    """Walk the steps one by one and stop at the first that fails."""
    parts = urlsplit(url)
    host = parts.hostname or ""
    port = parts.port or (443 if parts.scheme == "https" else 80)
    path = parts.path or "/"

    print(f"1. DNS      {host}")
    try:
        found = socket.getaddrinfo(host, port, type=socket.SOCK_STREAM)
    except socket.gaierror as error:
        print(f"   FAILED   the name does not resolve: {error}")
        return False
    print(f"   ok       {', '.join(str(info[4][0]) for info in found)}")

    print(f"2. CONNECT port {port}")
    try:
        socket.create_connection((host, port), timeout=5).close()
    except OSError as error:
        print(f"   FAILED   nobody answers on this port: {error}")
        return False
    print("   ok")

    step = "TLS + HTTP" if parts.scheme == "https" else "HTTP"
    print(f"3. {step}  GET {path}")
    connection_type = HTTPSConnection if parts.scheme == "https" else HTTPConnection
    connection = connection_type(host, port, timeout=10)
    try:
        connection.request("GET", path)
        response = connection.getresponse()
    except OSError as error:
        print(f"   FAILED   {error}")
        return False
    finally:
        connection.close()
    print(f"   answer   {response.status} {response.reason}")
    return response.status < 400


if __name__ == "__main__":
    sys.exit(0 if probe(sys.argv[1]) else 1)
