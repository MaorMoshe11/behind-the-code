בתיקייה שלכם הכול ירוק, גם עם --locked: uv run pytest כבר עדכן את uv.lock על הדיסק, ו---locked בודק שהקובץ על הדיסק תואם. הוא תואם. מה שלא תואם הוא מה שב-git, ואת זה רק עותק נקי רואה. ב-ci-test, uv run --locked סירב עוד לפני שהסקריפט התחיל; ב-GitHub הצעד Install the locked packages נכשל עם אותה הודעה.

הרמז: M uv.lock ב-git status. קובץ ששונה ולא נכנס ל-commit אחרי שינוי בתלויות הוא כמעט תמיד טעות. ההרגל שמונע את זה: uv add במקום עריכה ביד (הוא מעדכן את שני הקבצים יחד), ו-git status לפני כל commit. וזו התשובה לשאלה "למה CI אם אני מריץ הכול אצלי": אתם בודקים את התיקייה; CI בודק את ה-commit.

scripts\release_notes.py
"""Print the notes of one release from CHANGELOG.md, after checking that the tag is right.

    uv run python scripts/release_notes.py v0.9.0

Exits with an error when the tag does not match the version in pyproject.toml, or when
CHANGELOG.md has no section for it: then the release workflow stops before it builds.
"""

import argparse
import re
import sys
import tomllib
from pathlib import Path

ROOT = Path(__file__).resolve().parent.parent


def project_version() -> str:
    with (ROOT / "pyproject.toml").open("rb") as file:
        return str(tomllib.load(file)["project"]["version"])


def notes(version: str, changelog: str) -> str | None:
    """The text under '## [version]', up to the next '## ' heading."""
    pattern = rf"^## \[{re.escape(version)}\][^\n]*\n(.*?)(?=^## |\Z)"
    found = re.search(pattern, changelog, re.MULTILINE | re.DOTALL)
    return found.group(1).strip() if found else None


def main() -> None:
    parser = argparse.ArgumentParser(description="Check a release tag and print its notes.")
    parser.add_argument("tag", help="the git tag, for example v0.9.0")
    tag = parser.parse_args().tag
    version = project_version()
    if tag != f"v{version}":
        sys.exit(f"The tag {tag} does not match the version in pyproject.toml ({version}).")
    text = notes(version, (ROOT / "CHANGELOG.md").read_text(encoding="utf-8"))
    if not text:
        sys.exit(f"CHANGELOG.md has no notes for {version}.")
    print(text)


if __name__ == "__main__":
    main()

"""The release check: run the script as the release workflow does."""

import subprocess
import sys
from pathlib import Path

from agent import project_version

ROOT = Path(__file__).resolve().parent.parent


def release_notes(tag):
    return subprocess.run(
        [sys.executable, "scripts/release_notes.py", tag],
        cwd=ROOT,
        capture_output=True,
        text=True,
    )


def test_the_current_version_has_release_notes():
    # Fails as soon as pyproject.toml gets a new version without a CHANGELOG.md section.
    result = release_notes(f"v{project_version()}")
    assert result.returncode == 0, result.stderr
    assert "### " in result.stdout


def test_a_wrong_tag_stops_the_release():
    result = release_notes("v0.0.1")
    assert result.returncode == 1
    assert "does not match" in result.stderr
