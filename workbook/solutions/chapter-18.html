3כלי חדש, בלי לגעת בלולאה

agent\clock.py
Zone = Literal["UTC", "Asia/Jerusalem", "Europe/London", "America/New_York"]


class CurrentTimeInput(ToolInput):
    timezone: Zone = Field("UTC", description="Where the user is. Leave it out for UTC.")


def make_current_time(now: Callable[[], datetime] = lambda: datetime.now(UTC)) -> Tool:
    """The tool. A test passes its own `now`, so the answer is always the same."""

    def run(context: ToolContext, args: CurrentTimeInput) -> str:
        local = now().astimezone(ZoneInfo(args.timezone))
        return local.strftime(f"%A %Y-%m-%d %H:%M ({args.timezone})")

    return Tool(
        name="current_time",
        description=(
            "Today's date, the day of the week and the time. Call it when the answer depends "
            "on today: 'what day is it', 'how long until Monday', 'is it late in London'."
        ),
        input_model=CurrentTimeInput,
        run=run,
    )

ועוד שתי שורות: registry.register(make_current_time()) ב-build_registry(), ובמוק TIME = re.compile(r"\bwhat (?:time|day|date)\b", re.IGNORECASE) עם תנאי ב-wanted_calls. ב-tests\test_clock.py שלוש בדיקות: "What day is it?" עם שעון קבוע (Thursday 2026-11-19 21:30 (UTC)), אזור מהרשימה (Asia/Jerusalem נותן 23:30) ואזור שלא ממנה (Mars/Base, שגיאה עם הרשימה), ושהכלי רשום וקורא בלבד.

5resource בשרת ה-MCP

החתימות מהחבילה: on_list_resources מקבל (ctx, PaginatedRequestParams | None) ומחזיר ListResourcesResult; on_read_resource מקבל (ctx, ReadResourceRequestParams) ומחזיר ReadResourceResult.

agent\mcp_server.py (inside build_server; the types of ctx and params shortened)
    async def list_resources(ctx, params) -> types.ListResourcesResult:
        notes = types.Resource(
            uri=NOTES_URI,  # "notes://conversation"
            name="notes",
            description="Every note of this conversation, one per line (at most 20, oldest first).",
            mime_type="text/plain",
        )
        return types.ListResourcesResult(resources=[notes])

    async def read_resource(ctx, params) -> types.ReadResourceResult:
        if params.uri != NOTES_URI:
            raise MCPError(INVALID_PARAMS, f"No resource {params.uri!r}. There is {NOTES_URI}.")
        # The same code as the tool: one way to read notes, with its limits and its errors.
        outcome = await anyio.to_thread.run_sync(toolbox.call, "list_notes", {})
        text = types.TextResourceContents(uri=NOTES_URI, mime_type="text/plain", text=outcome.content)
        return types.ReadResourceResult(contents=[text])

ושני ארגומנטים ל-Server(...): on_list_resources=list_resources, on_read_resource=read_resource. ב-mcp_try.py, client.list_resources() אחרי הכלים, ו-read <uri> קורא ל-client.read_resource(uri). הבדיקה ב-test_tools.py קוראת את ה-resource דרך Client(server).

There is notes://conversation. אצל הלקוח. כלי או resource? כלי: המודל מחליט, באמצע ריצה, אם ומתי לקרוא, ועם איזה חיפוש. Resource: ה-host או המשתמש מחליטים לצרף את הנתונים להקשר מראש, למשל "צרף את ההערות שלי" לפני שאלה. Resource מתאים לנתונים קטנים שתמיד רלוונטיים; כלי מתאים כשיש הרבה נתונים ורק המודל יודע מה מהם צריך.
